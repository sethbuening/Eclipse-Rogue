# limb_tween_archive.gd
# Archived procedural limb rotation system — not currently in use.
# Re-integrate into player.gd by:
#   1. Adding the tween variables back (hand_tween_right/left, foot_tween_right/left)
#   2. Calling _tween_hands_to(value) and _tween_feet_to(value) from the direction setter
#   3. Replacing the instant position assignments in _ready() with the tween-initialised ones
#   4. Pasting the functions below back into player.gd

# ---- variables to restore ----
var hand_tween_right: Tween = null
var hand_tween_left:  Tween = null
var foot_tween_right: Tween = null
var foot_tween_left:  Tween = null
var hand_orbit_radius

# ---- call from direction setter instead of _snap_limbs_to() ----
func _tween_hands_to(dir: Vector2i) -> void:
	var to_right: Vector2 = _limb_orbit_pos(dir, hand_orbit_radius, false)
	var to_left:  Vector2 = _limb_orbit_pos(dir, hand_orbit_radius, true)
	var sweep:    float   = _limb_sweep_direction(hand_right_base, to_right)
	hand_tween_right = _tween_limb(hand_tween_right, hand_right_base, to_right, sweep,
		func(b: Vector2) -> void: hand_right_base = b)
	hand_tween_left  = _tween_limb(hand_tween_left,  hand_left_base,  to_left,  sweep,
		func(b: Vector2) -> void: hand_left_base  = b)

func _tween_feet_to(dir: Vector2i) -> void:
	var to_right: Vector2 = _limb_orbit_pos(dir, foot_orbit_radius, false)
	var to_left:  Vector2 = _limb_orbit_pos(dir, foot_orbit_radius, true)
	var sweep:    float   = _limb_sweep_direction(foot_right_base, to_right)
	foot_tween_right = _tween_limb(foot_tween_right, foot_right_base, to_right, sweep,
		func(b: Vector2) -> void: foot_right_base = b)
	foot_tween_left  = _tween_limb(foot_tween_left,  foot_left_base,  to_left,  sweep,
		func(b: Vector2) -> void: foot_left_base  = b)

# Returns the signed arc the right limb should travel; left limb reuses the same sign.
# angle_difference returns the shortest signed arc.
# If zero (already there), returns 0.0; ambiguous PI case defaults to clockwise.
func _limb_sweep_direction(from: Vector2, to: Vector2) -> float:
	var delta: float = angle_difference(from.angle(), to.angle())
	if is_zero_approx(delta):
		return 0.0
	return delta

func _tween_limb(tween: Tween, from: Vector2, to: Vector2, sweep: float, set_base: Callable) -> Tween:
	if tween:
		tween.kill()

	var from_angle:  float = from.angle()
	var from_radius: float = from.length()
	var to_radius:   float = to.length()

	# Reconstruct a signed delta for this specific limb using the shared sweep sign.
	# If the raw shortest arc goes the wrong way, add a full turn in sweep's direction
	# so both limbs always rotate together and never cross.
	var raw_delta:   float = angle_difference(from_angle, to.angle())
	var delta_angle: float
	if is_zero_approx(sweep):
		delta_angle = 0.0
	elif sign(raw_delta) == sign(sweep):
		delta_angle = raw_delta
	else:
		delta_angle = raw_delta + TAU * sign(sweep)

	tween = create_tween()
	tween.set_ease(Tween.EASE_OUT)
	tween.set_trans(Tween.TRANS_CUBIC)
	tween.tween_method(
		func(t: float) -> void:
			var angle:   float   = from_angle + delta_angle * t
			var radius:  float   = lerpf(from_radius, to_radius, t)
			var arc_pos: Vector2 = Vector2(cos(angle), sin(angle)) * radius
			# Y squish makes the arc read as a 3-D arc on a 2-D screen.
			# LIMB_ARC_Y_SQUISH = 0.75 is a good default.
			var squish:  float   = 1.0 - (1.0 - LIMB_ARC_Y_SQUISH) * sin(t * PI)
			set_base.call(Vector2(arc_pos.x, arc_pos.y * squish)),
		0.0, 1.0, LIMB_TWEEN_TIME
	)
	return tween
